## Example Code

Here's an example showing English Python code with Legesher:

```python
# Example: factorial function with error handling

from typing import Optional

def factorial(n):
    """Calculate the factorial of n."""
    if n < 0:
        raise valueerror("Negative numbers are not allowed")
    elif n == 0 or n == 1:
        return 1
    else:
        return n * factorial(n - 1)

def main():
    numbers = [int(x) for x in range(6)]

    for num in numbers:
        result = factorial(num)
        print(f"{num}! = {result}")

    # Demonstrate a while loop
    count = 0
    while count < 3:
        print(f"Count: {count}")
        count += 1

    # Demonstrate try/except
    try:
        factorial(-1)
    except valueerror as error:
        print(f"Error: {error}")
    finally:
        print("Done!")

if __name__ == "__main__":
    main()
```
